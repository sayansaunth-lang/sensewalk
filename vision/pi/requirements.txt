# Raspberry Pi vision pipeline dependencies (C2 track — see docs/TEAM.md).
# Install inside a venv, never into system Python:
#   python3 -m venv venv && source venv/bin/activate && pip install -r vision/pi/requirements.txt
#
# picamera2 is normally preinstalled on Raspberry Pi OS via apt
# (python3-picamera2) rather than pip — pip install is only a dev-machine
# fallback and may not have working libcamera bindings off-Pi.

opencv-python-headless>=4.9,<5
numpy>=1.26,<3
pytesseract>=0.3.10,<1
pyserial>=3.5,<4
pyttsx3>=2.90,<3   # offline speech; also needs `sudo apt install espeak-ng` on the Pi

# TFLite runtime: REQUIRED for the default detector (detector.py's TFLiteSSDDetector,
# the lightest option for a Raspberry Pi 3B with 1GB RAM — see docs/PI3B_LOW_RAM_SETUP.md).
# Not listed as a normal requirement because the right package depends on your OS and
# Python version; scripts/pi_setup.sh tries these in order:
#   pip install tflite-runtime      (piwheels hosts a prebuilt wheel on Raspberry Pi OS)
#   pip install ai-edge-litert      (Google's current packaging of the same runtime)
# Full `tensorflow` also works but is far too heavy for 1GB of RAM.
# (--detector-backend opencv-dnn needs no extra package but needs Caffe weights you must source yourself.)
